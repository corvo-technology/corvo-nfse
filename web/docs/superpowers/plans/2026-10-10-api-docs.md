## Global Constraints

- Package manager: **yarn** only (`yarn typecheck`, `yarn lint`, `yarn test`, `yarn build`).
- All visual styling uses existing tokens/classes only: `bg`, `bg-surface`, `bg-surface-alt`, `border-border`, `border-border-soft`, `text-text`, `text-text-2`, `text-text-3`, `accent`, `danger`, `warn`, `ok`, `font-mono`, `.banner-warn`, `.banner-accent`, `.banner-danger`. **No new colors, no hardcoded hex.**
- Docs content and chrome are bilingual; PT (`pt.ts`) is canonical, EN (`en.ts`) is the translation. Both typed against the same interfaces — structural drift is a compile error.
- **API-key auth only.** No portal/session content (magic-link, CSRF, cookies, `credentials: 'include'`) anywhere in the docs.
- Pages are server components. Client components are exactly: `CodeWindow` (tabs + copy) and `DocsSidebar` (mobile drawer + active state). Both files start with `"use client";`.
- Every inline-text string supports `` `backtick` `` code spans (rendered as `<code>`).
- Endpoint `scopes` use full scope names: `full_access`, `invoices_issue`, `invoices_view`, `invoices_read_write`.
- Code examples always use env vars `$BASE_URL` / `process.env.CORVO_API_KEY` — never real domains or keys.
- Endpoint JSON in examples must match the reference's envelopes exactly (`{ "data": ... }` vs `{ "props": ... }` vs bare arrays — see reference §3).

```ts
// lib/docs/types.ts
export type DocsLocale = "en" | "pt";
export const DOC_LOCALES: readonly DocsLocale[] = ["en", "pt"];

export type DocGroupKey = "gettingStarted" | "reference" | "guides";

export type DocSlug =
  | "authentication"
  | "errors"
  | "rate-limits"
  | "companies"
  | "certificates"
  | "invoices"
  | "webhooks"
  | "credits"
  | "billing"
  | "async-issuance"
  | "pagination"
  | "receiving-webhooks"
  | "integration-checklist";

export type HttpMethod = "GET" | "POST" | "PATCH" | "DELETE";

export interface TextBlock {
  kind: "text";
  body: string; // supports `code` spans
}

export interface ListBlock {
  kind: "list";
  items: string[]; // each item supports `code` spans
}

export interface TableBlock {
  kind: "table";
  head: string[];
  rows: string[][];
}

export interface CalloutBlock {
  kind: "callout";
  tone: "warn" | "accent" | "danger";
  body: string; // supports `code` spans
}

export interface HeadingBlock {
  kind: "heading";
  level: 2 | 3;
  body: string;
}

export interface CodeBlock {
  kind: "code";
  label?: string;
  language: "bash" | "json" | "js";
  code: string;
}

export interface ParamRow {
  name: string;
  type: string;
  required: boolean;
  description: string; // supports `code` spans
}

export interface EndpointBlock {
  kind: "endpoint";
  method: HttpMethod;
  path: string;
  scopes: string[];
  success: string; // e.g. "201 Created"
  summary: string; // supports `code` spans
  params?: { title?: string; rows: ParamRow[] };
  examples: { curl: string; node: string };
  response: { label: string; json: string };
  notes?: CalloutBlock[];
}

export type DocBlock =
  | TextBlock
  | ListBlock
  | TableBlock
  | CalloutBlock
  | HeadingBlock
  | CodeBlock
  | EndpointBlock;

export interface DocSection {
  title: string;
  description?: string;
  blocks: DocBlock[];
}

export interface DocsChrome {
  homeLabel: string;
  groups: Record<DocGroupKey, string>;
  copy: {
    copy: string;
    copied: string;
    response: string;
    close: string;
    menu: string;
    table: { field: string; type: string; required: string; description: string };
  };
  localeSwitch: Record<DocsLocale, string>;
  notFound: { title: string; body: string };
}

export interface DocsTree {
  chrome: DocsChrome;
  home: DocSection;
  pages: Record<DocSlug, DocSection>;
}
```

```ts
// lib/docs/registry.ts
import { en } from "./content/en";
import { pt } from "./content/pt";
import type { DocGroupKey, DocsLocale, DocsTree } from "./types";

export const DOC_GROUPS: { key: DocGroupKey; slugs: import("./types").DocSlug[] }[] = [
  { key: "gettingStarted", slugs: ["authentication", "errors", "rate-limits"] },
  {
    key: "reference",
    slugs: ["companies", "certificates", "invoices", "webhooks", "credits", "billing"],
  },
  {
    key: "guides",
    slugs: ["async-issuance", "pagination", "receiving-webhooks", "integration-checklist"],
  },
];

export const DOC_SLUGS = DOC_GROUPS.flatMap((g) => g.slugs);

const TREES: Record<DocsLocale, DocsTree> = { en, pt };

export function getDocsTree(locale: string): DocsTree | null {
  return locale === "en" || locale === "pt" ? TREES[locale] : null;
}
```

**Transcription rules (apply everywhere):**
1. Every endpoint becomes an `EndpointBlock` with `method`, `path` (include `/v1` prefix), `scopes` (full names — map F→`full_access`, I→`invoices_issue`, V→`invoices_view`, RW→`invoices_read_write`), `success` (e.g. `"201 Created"`), `summary` (one sentence from the reference's function description).
2. Request fields from the reference's per-endpoint tables become `params.rows` (`required: true` when the reference says Obrigatório/Sim). Reference's "Regra" column → `description` (keep `` `code` `` around literals like field values).
3. Every endpoint gets BOTH a `curl` and a `node` example. `curl` is transcribed/adapted from the reference; `node` mirrors it with `fetch`. Template:

```
curl: `curl -X POST "$BASE_URL/companies" \\
  -H "X-API-Key: $CORVO_API_KEY" \\
  -H "Content-Type: application/json" \\
  -d '{ ...body... }'`

node: `const res = await fetch(\`\${BASE_URL}/companies\`, {
  method: "POST",
  headers: {
    "X-API-Key": process.env.CORVO_API_KEY,
    "Content-Type": "application/json",
  },
  body: JSON.stringify({ ...same body... }),
});
const body = await res.json();
console.log(body.data.id); // { data: Company }`
```

4. `response.json` = a trimmed illustrative JSON from the reference (keep real field names/envelope; `"..."` to elide long lists is allowed).
5. Errors/warnings from the reference's prose ("retorna 409", "422 por plano bloqueado") become `notes` callouts (`tone: "warn"` for errors, `"accent"` for tips/notes, `"danger"` for destructive/security).
6. GET endpoints with query params (e.g. `limit`, `cursor`) get a `params` table with `required: false` and the reference's defaults.
7. Pagination quirk notes (companies `limit + 1` behavior, `createdAt|id` cursor) go on the GET list endpoints as `notes` — and the dedicated `pagination` guide (Task 4) links conceptually but keeps its own full explanation.

```tsx
const CODE_CLASS =
  "rounded bg-surface px-1 py-0.5 font-mono text-[0.85em] text-accent";

function renderParts(body: string) {
  return body.split(/`([^`]+)`/g).map((part, i) =>
    i % 2 === 1 ? (
      <code key={i} className={CODE_CLASS}>
        {part}
      </code>
    ) : (
      part
    ),
  );
}

export function InlineText({ body }: { body: string }) {
  return <>{renderParts(body)}</>;
```

```tsx
"use client";
import { useState } from "react";

type Tab = "curl" | "node";

export function CodeWindow({
  curl,
  node,
  copyLabel,
  copiedLabel,
}: {
  curl: string;
  node: string;
  copyLabel: string;
  copiedLabel: string;
}) {
  const [tab, setTab] = useState<Tab>("curl");
  const [copied, setCopied] = useState(false);
  const code = tab === "curl" ? curl : node;

  async function copy() {
    await navigator.clipboard.writeText(code);
    setCopied(true);
    setTimeout(() => setCopied(false), 1500);
  }

  const tabClass = (active: boolean) =>
    `rounded-md px-2 py-1 text-xs font-mono transition-colors ${
      active ? "bg-surface-alt text-text" : "text-text-4 hover:text-text-2"
    }`;

  return (
    <div className="overflow-hidden rounded-xl border border-border bg-surface">
      <div className="flex items-center justify-between border-b border-border px-3 py-2">
        <div className="flex items-center gap-2">
          <span className="size-2.5 rounded-full bg-[#ff5f57]" aria-hidden="true" />
          <span className="size-2.5 rounded-full bg-[#febc2e]" aria-hidden="true" />
          <span className="size-2.5 rounded-full bg-[#28c840]" aria-hidden="true" />
        </div>
        <div className="flex items-center gap-1" role="tablist" aria-label="Example language">
          <button type="button" role="tab" aria-selected={tab === "curl"} onClick={() => setTab("curl")} className={tabClass(tab === "curl")}>
            cURL
          </button>
          <button type="button" role="tab" aria-selected={tab === "node"} onClick={() => setTab("node")} className={tabClass(tab === "node")}>
            JavaScript
          </button>
        </div>

          onClick={copy}
```

```tsx
// components/docs/doc-blocks.tsx
import type { DocBlock, DocsChrome } from "@/lib/docs/types";
import { Callout } from "./callout";
import { EndpointCard } from "./endpoint-card";
import { InlineText } from "./inline-text";

const CODE_BLOCK_LANG_LABEL = { bash: "bash", json: "json", js: "javascript" } as const;

  blocks,
  copy,
}: {
  blocks: DocBlock[];
  copy: DocsChrome["copy"];
}) {
  return (
    <div className="space-y-6">
      {blocks.map((block, i) => {
        switch (block.kind) {
          case "heading":
            return block.level === 2 ? (
              <h2 key={i} className="pt-4 text-xl font-semibold text-text">
                <InlineText body={block.body} />
              </h2>
            ) : (
              <h3 key={i} className="pt-2 text-lg font-medium text-text">

            return (

              </p>
            );
          case "list":
            return (
              <ul key={i} className="list-disc space-y-1.5 pl-5 text-sm text-text-2">
                {block.items.map((item, j) => (
                  <li key={j}>
                    <InlineText body={item} />
                  </li>
                ))}
              </ul>
            );
          case "table":
            return (
              <div key={i} className="overflow-x-auto rounded-xl border border-border">
                <table className="w-full text-left text-sm">
                  <thead className="bg-surface">
                    <tr className="text-text-3">
                      {block.head.map((h, j) => (
                        <th key={j} className="p-3 font-medium">
                          {h}
                        </th>
                      ))}
                    </tr>
                  </thead>
                  <tbody>
                    {block.rows.map((row, j) => (
                      <tr key={j} className="border-t border-border align-top">
                        {row.map((cell, k) => (
                          <td key={k} className="p-3 text-text-2">
                            <InlineText body={cell} />
                          </td>
                        ))}
                      </tr>
                    ))}
                  </tbody>
                </table>
              </div>
            );
          case "callout":
            return <Callout key={i} tone={block.tone} body={block.body} />;
          case "code":
            return (
              <div key={i}>
                <p className="mb-1.5 font-mono text-[11px] uppercase tracking-wide text-text-4">
                  {block.label ?? CODE_BLOCK_LANG_LABEL[block.language]}

  );
```
